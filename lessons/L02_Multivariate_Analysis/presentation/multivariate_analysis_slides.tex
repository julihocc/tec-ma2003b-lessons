\begin{frame}{Random Vectors \& Population Parameters}
Let
\(\mathbf{X} = \left\lbrack X_{1},X_{2},\ldots,X_{p} \right\rbrack^{T} \in \mathbb{R}^{p}\)
be a \(p\)-dimensional random vector.

\textbf{Population Mean Vector:}
\[\mathbf{\mu} = \mathbb{E}\left\lbrack \mathbf{X} \right\rbrack = \begin{bmatrix}
\mathbb{E}\left\lbrack X_{1} \right\rbrack & \mathbb{E}\left\lbrack X_{2} \right\rbrack & \ldots & \mathbb{E}\left\lbrack X_{p} \right\rbrack
\end{bmatrix}^{T}\]

\textbf{Population Covariance Matrix:}
\[\mathbf{\Sigma} = \operatorname{Cov}(\mathbf{X}) = \mathbb{E}\left\lbrack \left( \mathbf{X} - \mathbf{\mu} \right)\left( \mathbf{X} - \mathbf{\mu} \right)^{T} \right\rbrack \in \mathbb{R}^{p \times p}\]

\textbf{Property:} \(\mathbf{\Sigma}\) is always symmetric and positive
semi-definite (\(\mathbf{\Sigma} \succeq 0\)).
\end{frame}

\begin{frame}{Sample Statistics \& Geometric Interpretation}
Given data matrix \(\mathbf{X} \in \mathbb{R}^{n \times p}\):

\textbf{Sample Mean \& Covariance Matrix:}
\[\overline{\mathbf{x}} = \frac{1}{n}\mathbf{X}^{T}\mathbf{1}_{n},\quad\quad\mathbf{S} = \frac{1}{n - 1}\sum_{i = 1}^{n}\left( \mathbf{x}_{i} - \overline{\mathbf{x}} \right)\left( \mathbf{x}_{i} - \overline{\mathbf{x}} \right)^{T}\]

\textbf{Sample Correlation Matrix:}
\[\mathbf{R} = \mathbf{D}_{\mathbf{S}}^{- \frac{1}{2}}\mathbf{S}\mathbf{D}_{\mathbf{S}}^{- \frac{1}{2}},\quad r_{jk} = \frac{s_{jk}}{s_{j}s_{k}} = \cos(\theta_{jk})\]

The correlation \(r_{jk}\) is the cosine of the angle between
mean-centered observation vectors in sample space \(\mathbb{R}^{n}\).
\end{frame}

\begin{frame}{The Multivariate Normal Distribution (MVN)}
\(\mathbf{X} \sim \mathcal{N}_{p\left( \mathbf{\mu},\mathbf{\Sigma} \right)}\)
has the joint probability density function:

\[f\left( \mathbf{x} \right) = \frac{1}{(2\pi)^{\frac{p}{2}}|\mathbf{\Sigma}|^{\frac{1}{2}}}\exp( - \frac{1}{2}\left( \mathbf{x} - \mathbf{\mu} \right)^{T}\mathbf{\Sigma}^{- 1}\left( \mathbf{x} - \mathbf{\mu} \right))\]

\textbf{Fundamental Properties:}

\begin{itemize}
\item
  Any linear combination \(\mathbf{a}^{T}\mathbf{X}\) is univariate
  normal
  \(\mathcal{N}(\mathbf{a}^{T}\mathbf{\mu},\mathbf{a}^{T}\mathbf{\Sigma}\mathbf{a})\).
\item
  All marginals and conditionals are multivariate normal.
\item
  \(\operatorname{Cov}(X_{j},X_{k}) = 0 \Longleftrightarrow X_{j}\text{  and  }X_{k}\text{  are statistically independent}\).
\end{itemize}
\end{frame}

\begin{frame}{Geometry: Contour Ellipsoids \& Mahalanobis Distance}
\textbf{Squared Mahalanobis Distance:}
\[D^{2}\left( \mathbf{x},\mathbf{\mu} \right) = \left( \mathbf{x} - \mathbf{\mu} \right)^{T}\mathbf{\Sigma}^{- 1}\left( \mathbf{x} - \mathbf{\mu} \right)\]

\textbf{Contour Surfaces:} Surfaces of constant probability density form
hyper-ellipsoids:
\[\{\mathbf{x}:\left( \mathbf{x} - \mathbf{\mu} \right)^{T}\mathbf{\Sigma}^{- 1}\left( \mathbf{x} - \mathbf{\mu} \right) = c^{2}\}\]

\textbf{Chi-Square Distribution Property:}
\[D^{2}\left( \mathbf{X},\mathbf{\mu} \right) \sim \chi_{p}^{2}\]
Enables empirical testing of multivariate normality via
\textbf{Chi-Square Q-Q Plots}.
\end{frame}

\begin{frame}{Fisher \(z\)-Transformation for Correlations}
\textbf{Problem:} Sampling distribution of \(r\) is skewed when
\(|\rho| > 0\).

\textbf{Fisher's Transformation:}
\[z = \operatorname{arctanh}(r) = \frac{1}{2}\ln(\frac{1 + r}{1 - r}) \sim^{\text{approx}}\mathcal{N}(\operatorname{arctanh}(\rho),\frac{1}{n - 3})\]

\textbf{95\% Confidence Interval for \(\rho\):}
\[\left\lbrack \rho_{L},\rho_{U} \right\rbrack = \left\lbrack \tanh(z - 1.96\frac{1}{\sqrt{n - 3}}),,\tanh(z + 1.96\frac{1}{\sqrt{n - 3}}) \right\rbrack\]
\end{frame}
