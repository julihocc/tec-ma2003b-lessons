\textbf{Goal:} Construct discriminant score functions
\(d_{i\left( \mathbf{x} \right)}\) that maximize separation between
groups while minimizing within-group scatter.

\begin{frame}{Fisher's Linear Discriminant (Two Populations)}
Let \(\mathbf{x} \in \mathbb{R}^{p}\) from population
\(\Pi_{1} \sim \left( \mathbf{\mu}_{1},\mathbf{\Sigma} \right)\) or
\(\Pi_{2} \sim \left( \mathbf{\mu}_{2},\mathbf{\Sigma} \right)\) with
shared covariance \(\mathbf{\Sigma}\).

\textbf{Fisher's Criterion:} Find linear combination
\(y = \mathbf{a}^{T}\mathbf{x}\) maximizing the ratio of between-group
to within-group variance:
\[J\left( \mathbf{a} \right) = \frac{\left( \mathbf{a}^{T}\mathbf{\mu}_{1} - \mathbf{a}^{T}\mathbf{\mu}_{2} \right)^{2}}{\mathbf{a}^{T}\mathbf{\Sigma}\mathbf{a}}\]

\textbf{Optimal Coefficient Vector:}
\[\mathbf{a}^{\ast} = \mathbf{\Sigma}^{- 1}\left( \mathbf{\mu}_{1} - \mathbf{\mu}_{2} \right)\quad\left( \text{sample estimator:  }\hat{\mathbf{a}} = \mathbf{S}_{\text{pooled}}^{- 1}\left( {\overline{\mathbf{x}}}_{1} - {\overline{\mathbf{x}}}_{2} \right) \right)\]

\textbf{Decision Rule (Equal Priors):} Assign to \(\Pi_{1}\) if
\(\mathbf{a}^{T}\mathbf{x} \geq \frac{1}{2}\mathbf{a}^{T\left( \mathbf{\mu}_{1} + \mathbf{\mu}_{2} \right)} = m\);
otherwise assign to \(\Pi_{2}\).
\end{frame}

\begin{frame}{Prior Probabilities \& Misclassification Costs (ECM)}
In real business applications, groups have unequal base rates
\(p_{1},p_{2}\) and misclassification penalties
\(C\left( 2|1 \right) \neq C\left( 1|2 \right)\).

\textbf{Expected Cost of Misclassification (ECM):}
\[\operatorname{ECM} = C\left( 2|1 \right)p_{1}P\left( 2|1 \right) + C\left( 1|2 \right)p_{2}P\left( 1|2 \right)\]

\textbf{Bayes Optimal Decision Rule (Minimize ECM):} Assign
\(\mathbf{x}\) to \(\Pi_{1}\) if:
\[\frac{f_{1}\left( \mathbf{x} \right)}{f_{2}\left( \mathbf{x} \right)} \geq \left( \frac{C\left( 1|2 \right)}{C\left( 2|1 \right)} \right)\left( \frac{p_{2}}{p_{1}} \right)\]

Taking logarithms under multivariate normality yields an adjusted linear
cutoff:
\[d_{L\left( \mathbf{x} \right)} = \left( \mathbf{\mu}_{1} - \mathbf{\mu}_{2} \right)^{T}\mathbf{\Sigma}^{- 1}\mathbf{x} - {\frac{1}{2}(\mathbf{\mu}_{1} - \mathbf{\mu}_{2})}^{T}\mathbf{\Sigma}^{- 1}\left( \mathbf{\mu}_{1} + \mathbf{\mu}_{2} \right) \geq \ln\left\lbrack \frac{C\left( 1|2 \right)p_{2}}{C\left( 2|1 \right)p_{1}} \right\rbrack\]
\end{frame}

\begin{frame}{Multi-Group Discrimination (\(k > 2\) Populations)}
When classifying among \(k\) groups, we define between-group matrix
\(\mathbf{B}\) and within-group matrix \(\mathbf{W}\):
\[\mathbf{W} = \sum_{i = 1}^{k}\sum_{j = 1}^{n_{i}}\left( \mathbf{x}_{ij} - {\overline{\mathbf{x}}}_{i} \right)\left( \mathbf{x}_{ij} - {\overline{\mathbf{x}}}_{i} \right)^{T},\quad\mathbf{B} = \sum_{i = 1}^{k}n_{i}\left( {\overline{\mathbf{x}}}_{i} - \overline{\mathbf{x}} \right)\left( {\overline{\mathbf{x}}}_{i} - \overline{\mathbf{x}} \right)^{T}\]

\textbf{Canonical Discriminant Functions:} Eigenvalue problem:
\(\mathbf{W}^{- 1}\mathbf{B}\mathbf{v}_{r} = \lambda_{r}\mathbf{v}_{r}\).

\begin{itemize}
\item
  Number of non-zero canonical functions: \(s = \min(k - 1,p)\).
\item
  \(r\)-th canonical score: \(Z_{r} = \mathbf{v}_{r}^{T}\mathbf{x}\).
\item
  Canonical correlations:
  \(\rho_{r} = \sqrt{\frac{\lambda_{r}}{1 + \lambda_{r}}}\).
\end{itemize}
\end{frame}

\begin{frame}{Wilks' Lambda \& Stepwise Variable Selection}
\textbf{Wilks' Lambda (\(\Lambda\)):}
\[\Lambda = \frac{|\mathbf{W}|}{|\mathbf{W} + \mathbf{B}|} = \prod_{r = 1}^{s}\frac{1}{1 + \lambda_{r}},\quad\Lambda \in (0,1\rbrack\]

\begin{itemize}
\item
  \(\Lambda \rightarrow 0\): High group separation (predictors strongly
  discriminate).
\item
  \(\Lambda \rightarrow 1\): No group separation (group centroids are
  identical).
\end{itemize}

\textbf{Stepwise Variable Selection:}

\begin{itemize}
\item
  \textbf{Forward Selection:} Add predictor that produces largest
  reduction in \(\Lambda\) (\(F_{\text{to enter}} > F_{\text{in}}\)).
\item
  \textbf{Backward Elimination:} Remove least contributing predictor
  (\(F_{\text{to remove}} < F_{\text{out}}\)).
\item
  Prevents overfitting and maximizes model parsimony.
\end{itemize}
\end{frame}

Covariance Assumption & Equal:
\(\mathbf{\Sigma}_{1} = \mathbf{\Sigma}_{2} = \ldots = \mathbf{\Sigma}_{k} = \mathbf{\Sigma}\)
& Unequal: \(\mathbf{\Sigma}_{i} \neq \mathbf{\Sigma}_{j}\) per class \\

/ Parabolas) \\
Parameters to Estimate & \(p\) means + 1 pooled \(\mathbf{\Sigma}\)
(\(O\left( p^{2} \right)\)) & \(k\) means + \(k\) individual
\(\mathbf{\Sigma}_{i}\) (\(O\left( kp^{2} \right)\)) \\
Sample Size Requirement & Moderate (\(n > p\)) & Large per class
(\(n_{i} > p\)) \\
Robustness & More robust to small sample sizes & Captures complex
non-linear variance patterns \\

\begin{frame}{Summary \& Takeaways}
\begin{itemize}
\item
  \textbf{Discriminant Analysis} is a supervised classification
  technique grounded in multivariate probability distributions.
\item
  \textbf{LDA} assumes homogeneous covariance matrices and creates
  planar decision surfaces.
\item
  \textbf{QDA} accommodates class-specific covariance matrices at the
  cost of estimating more parameters.
\item
  Always evaluate misclassification costs and prior probabilities to
  minimize expected business loss.
\item
  Inspect canonical loadings and Wilks' Lambda to identify key
  discriminating features.
\end{itemize}
